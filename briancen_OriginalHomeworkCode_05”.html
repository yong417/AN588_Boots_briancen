# bootstrap start
for (i in 1:n_boot) {
  # each time we will take a sample with replacement to obtain sample from the population,

  # fit the sample data into the linear log transformed regression model
  boot_fit <- lm(log(HomeRange_km2) ~ log(Body_mass_female_mean), data = boot_sample)

  boot_coefs[i, ] <- coef(boot_fit)

bootstrap_function <- function(d, m, conf.level = 0.95, n = 1000) {
  # the input m is as characters and need to change it to formula to feed into the lm()
  model_formula <- as.formula(m)
  
  # fit the model into the liner regression model
  lm_model <- lm(model_formula, data = d)
  
  # Extract the slope and intercept coefficient of th model
  lm_coef <- coef(lm_model)
  
  # Retrieves the standard errors for the coefficients from the summary of the linear model
  lm_se <- summary(lm_model)$coefficients[, "Std. Error"]
  
  # calculate the confidence interval for the model
  lm_ci <- confint(lm_model, level = conf.level)
  
  # Define coefficient names from the full model estimates
  coef_names <- names(lm_coef)
  
  # create a matrix to store the bootstrap values
  boot_coefs <- matrix(NA, nrow = n, ncol = length(coef_names))
  colnames(boot_coefs) <- coef_names
  
  # set seed for reproducibility 
  set.seed(303)
  
  # Determine the sample size in th input data frame d
  N <- nrow(d)
  
  # use for loop like previous to 
  for (i in 1:n) {
      # get a sample from 1 to sample size with replacement
      boot_sample <- d[sample(1:N, size = N, replace = TRUE), ]
      
      # Fit linear model on the bootstrap sample.
      boot_model <- lm(model_formula, data = boot_sample)
      
      # Store the estimated coefficients in th  matrix we created fill them row by row
      boot_coefs[i, ] <- coef(boot_model)
    }
  
  # Compute the mean of the bootstrap coefficient estimates for each coefficient
  boot_mean <- apply(boot_coefs, 2, mean)
  
  # Compute the bootstrap standard errors as the standard deviation of the bootstrap estimates
  boot_se <- apply(boot_coefs, 2, sd)
  
  # Compute the bootstrap confidence intervals.
  boot_ci <- apply(boot_coefs, 2, quantile, probs = c((1 - conf.level)/2, 1 - (1 - conf.level)/2))
  
  # Combine the full-model and bootstrap results into a single data frame.
  result_df <- data.frame(
    Coefficient = coef_names,
    model_Estimate = lm_coef,
    model_SE = lm_se,
    model_CI_Lower = lm_ci[, 1],
    model_Upper = lm_ci[, 2],
    Bootstrap_Mean = boot_mean,
    Bootstrap_SE = boot_se,
    Bootstrap_CI_Lower = boot_ci[1, ],
    Bootstrap_CI_Upper = boot_ci[2, ]
  )
  
  # Return the resulting data frame so that the caller can review the estimates.
  return(result_df)
}
